### .env à mettre dans le dossier qui comprend tout les services

DB_SERVICE_URL=http://localhost:1999/

PAYMENT_SERVICE_URL=http://localhost:2005/
STRIPE_PUBLIC_KEY=
STRIPE_PRIVATE_KEY=

# Authentication
AUTH_SERVICE_PORT=2001
AUTH_SERVICE_URL=http://localhost:2001/
JWT_SECRET=
GOOGLE_CLIENT_ID=
GOOGLE_CLIENT_SECRET=
FACEBOOK_CLIENT_ID=
FACEBOOK_CLIENT_SECRET=
LINKEDIN_CLIENT_ID=
LINKEDIN_CLIENT_SECRET=

# Front

VITE_DB_SERVICE_URL=http://localhost:1999/

# Notifications

MAIL_HOST=

# IA
IA_SERVICE_PORT=2004

OPENAI_API_KEY=

### Script pout lancer tous les services rapidement

```sh
#!/bin/bash

echo "Lancement de tous les projets..."

for dir in */ ; do
  if [ -f "$dir/package.json" ]; then
    echo "➡️  Lancement de $dir"
    (cd "$dir" && npm run dev) &
  fi
done

wait
```
